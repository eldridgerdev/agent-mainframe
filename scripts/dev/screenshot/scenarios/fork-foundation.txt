# Fork a private repository with a real working change and a prompt-only hook.
# No harness is started; the hook prompt is cancelled after the stopped fork is registered.
run:set -e; FX="$SHOT_ROOT/fork-repo"; mkdir -p "$FX"; git -C "$FX" init -q -b main; git -C "$FX" config user.name Fixture; git -C "$FX" config user.email fixture@example.test; printf 'original\n' > "$FX/invoice.txt"; printf '{"lifecycle_hooks":{"on_worktree_created":{"script":"true","prompt":{"title":"Choose setup","options":["Continue"]}}}}\n' > "$FX/amf.json"; git -C "$FX" add .; git -C "$FX" commit -qm initial; git -C "$FX" checkout -qb invoice
run:printf '{"path":"%s/fork-repo","project_name":"fork-demo","dry_run":false}' "$SHOT_ROOT" > "$SHOT_ROOT/project.json"; "$AMF_BIN" automation create-project --file "$SHOT_ROOT/project.json"
run:printf '{"project_name":"fork-demo","branch":"invoice","use_worktree":false,"agent":"%s","mode":"vibe","review":false,"plan_mode":false,"create_terminal":false,"enable_chrome":false,"no_start":true,"dry_run":false}' "$CAPTURE_HARNESS_SLUG" > "$SHOT_ROOT/feature.json"; "$AMF_BIN" automation create-feature --file "$SHOT_ROOT/feature.json"
run:printf 'staged\n' > "$SHOT_ROOT/fork-repo/invoice.txt"; git -C "$SHOT_ROOT/fork-repo" add invoice.txt; printf 'working content\n' > "$SHOT_ROOT/fork-repo/invoice.txt"
run:tmux respawn-pane -k -t "$SESSION" "$AMF_BIN"
