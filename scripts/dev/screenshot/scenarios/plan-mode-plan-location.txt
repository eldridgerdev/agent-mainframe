# Proves that an accepted plan is written to the feature root and that a
# running Codex session receives the same unsent kickoff prompt as Claude.
#
# Requires --seed with a create-project payload for a throwaway repository.
# CODEX_SHOT_SEED_FEATURE must point to a create-feature payload for that
# project using the Codex harness, feature name `codex-plan-handoff`, branch
# `codex-plan-handoff`, and one session named `Codex 1`.
#
# The draft seed reopens the interview at its review gate, avoiding a real
# headless synthesis call. The scenario enables Codex in the isolated AMF
# registry before it creates the feature.

shot:dashboard-seeded

key:j
key:j
key:Enter

wait:600

run:$AMF_BIN automation create-feature --file "$CODEX_SHOT_SEED_FEATURE"
wait:1400
shot:dashboard-codex-feature

run:python3 "$SCRIPT_DIR/seed_plan_draft.py" "$XDG_CONFIG_HOME/amf/amf.db"
wait:500
key:j
wait:500
key:P
wait:900
key:r
wait:900

wait:1400

# the kickoff prompt. Clear the underlying terminal so the evidence focuses
