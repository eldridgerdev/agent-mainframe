# learning-mode: proves the Learning Mode overlay added on the
# learning-mode branch — the `K` entry key, the read-only three-pane
# surface, the first-open help, starter questions, the question prompt's
# spelled-out anchor, and a real headless answer rendered as markdown
# with the `F` follow-up affordance.
#
# Seeded with a small throwaway "taskline" repo (README.md, CLAUDE.md,
# Cargo.toml, src/main.rs) so the pinned "Start here" group has real
# files to find.
#
# Run with:
#   ./amf-capture.sh --scenario <this file> \
#     --seed <scratch>/seed-project.json \
#     --seed-feature <scratch>/seed-feature.json \
#     --geometry 160x44

wait:2000
shot:dashboard

# `K` on the selected project opens Learning Mode on its first feature.
# The project has never been opened in Learning Mode, so the help
# overlay shows itself unprompted — a newcomer's discovery path is this
# overlay, not the source.
key:K
wait:2500
shot:first-open-help

# Esc dismisses the intro and leaves the browsing surface behind it.
# The cursor opens on the "Tour this whole project" row rather than a
# group header, so the first thing on screen is a question you can ask.

# nothing on its own. The prompt states its anchor in plain words.
