# unsent-prompt-recovery: visual proof of the "unsent prompt" recovery
# mechanism (App::stash_lost_prompt, src/app/prompt_library.rs). When a
# prompt is computed to seed a session's composer -- e.g. starting a feature
# from a TODO -- but the launch that would have carried it fails (agent
# limit, harness spawn error), the prompt is no longer just discarded. It is
# saved two ways: as a real, editable User template in the Prompt Library
# (tagged "unsent", leader P) and as a row Latest Prompt recall (leader l)
# surfaces for that checkout once a session exists to view it from.
#
# Self-contained: seeds a project/feature/session plus the prompt_templates
# + unsent_prompts rows directly in the scratch database
# (seed-unsent-prompt-demo.py), the same way latest-prompt-menu.txt does --
# no real harness process, no live launch failure needed to demonstrate the
# recovery UI.
#
# Run with:
#   ./amf-capture.sh --scenario scenarios/unsent-prompt-recovery.txt

run:set -e; WORKDIR="$SHOT_ROOT/unsent-prompt-demo"; mkdir -p "$WORKDIR"; printf '{"path":"%s","project_name":"unsent-prompt-demo","dry_run":false}' "$WORKDIR" > "$SHOT_ROOT/project.json"; "$AMF_BIN" automation create-project --file "$SHOT_ROOT/project.json"

# Insert the feature + session, the auto-saved Prompt Library entry, and the
# Latest Prompt recall row -- all keyed to the same workdir, as
# App::stash_lost_prompt would produce on a failed TODO-spawn launch.
run:WORKDIR="$SHOT_ROOT/unsent-prompt-demo"; TMUX_SESSION="amf-unsent-prompt-demo"; python3 "$REPO_ROOT/scripts/dev/screenshot/seed-unsent-prompt-demo.py" "$XDG_CONFIG_HOME/amf/amf.db" "$WORKDIR" "$TMUX_SESSION"; if ! tmux has-session -t "$TMUX_SESSION" 2>/dev/null; then tmux new-session -d -s "$TMUX_SESSION" -n claude -c "$WORKDIR"; fi

# Automation and the direct DB seed above both ran after the TUI had
# already loaded, so restart it against the seeded scratch database before
# navigating.
run:tmux respawn-pane -k -t "$SESSION" "$AMF_BIN"
run:for i in {1..50}; do tmux capture-pane -p -t "$SESSION" > "$XDG_STATE_HOME/unsent-prompt-pane.txt"; grep -q "unsent-prompt-demo" "$XDG_STATE_HOME/unsent-prompt-pane.txt" && break; sleep 0.2; done
wait:2000
note:Dashboard with the seeded project and feature -- no failure has happened yet, this is just the setup for the two recovery surfaces below.
shot:dashboard

# Search reaches the session directly, including while its project tree is
