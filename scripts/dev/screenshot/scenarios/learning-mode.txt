# learning-mode: the Learning Mode overlay end to end — the `K` entry
# key, the read-only three-pane surface, the first-open help, starter
# questions, the question prompt's spelled-out anchor, a real headless
# answer rendered as markdown, a follow-up, and handing the answer to a
# live agent.
#
# The question deliberately asks for an answer shaped as a bulleted list
# whose every bullet *opens* with an inline code span: that is the shape
# the shared markdown renderer used to drop (`• — it worked` instead of
# "`Ok(())` — it worked"), so the answer frame doubles as the regression
# shot for that fix.
#
# Seed a small throwaway repo with README.md, CLAUDE.md, Cargo.toml and
# src/main.rs so the pinned "Start here" group has real files to find.
#
# Run with:
#   ./amf-capture.sh --scenario scenarios/learning-mode.txt \
#     --seed <scratch>/seed-project.json \
#     --seed-feature <scratch>/seed-feature.json \
#     --geometry 160x44

wait:2000
shot:dashboard

# `K` on the selected project opens Learning Mode on its first feature.
# The project has never been opened in Learning Mode, so the help shows
# itself unprompted — a newcomer's discovery path is this overlay, not
# the source.
key:K
wait:2500
shot:first-open-help

# Esc dismisses the intro and leaves the browsing surface behind it. The
# cursor opens on the "Tour this whole project" row rather than a group
# header, so the first thing on screen is a question you can ask.

# nothing on its own. The prompt states its anchor in plain words.
